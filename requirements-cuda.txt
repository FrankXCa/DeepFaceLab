# Install:
#   uv venv .venv --python 3.12
#   uv pip install --index-strategy unsafe-best-match \
#       -r requirements-cuda.txt -r requirements-dev.txt
# (pip users: plain `pip install -r` is fine; the uv flag is needed
# because the PyTorch index mirrors a few numpy versions and uv's
# default first-index-wins strategy would otherwise miss numpy on PyPI.)
#
